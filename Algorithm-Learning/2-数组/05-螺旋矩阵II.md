## 循环不变量原则
在循环中确保始终是**左闭右开**，或者是**左开右闭**。
![[引用文件/Algorithm-Learning/2-05-1.png|430]]
```cpp
class Solution {
public:
    vector<vector<int>> generateMatrix(int n) {
        vector<vector<int>> nums(n, vector<int>(n, 0));
        int num = 1;
        int loop = n/2;
        int start_row = 0;
        int start_column = 0;
        int len = n-1;
        while(loop--){
            for(int i=0; i<len; i++){
                nums[start_row][start_column+i] = num;
                num++;
            }
            for(int i=0; i<len; i++){
                nums[start_row+i][start_column+len] = num;
                num++;
            }
            for(int i=0; i<len; i++){
                nums[start_row+len][start_column+len-i] = num;
                num++;
            }
            for(int i=0; i<len; i++){
                nums[start_row+len-i][start_column] = num;
                num++;
            }
            start_row++;
            start_column++;
            len -= 2;
        }
        if(n%2){
            nums[n/2][n/2] = num; // 奇数矩阵单独处理中间的值。
        }
        return nums;
    }
};
```
