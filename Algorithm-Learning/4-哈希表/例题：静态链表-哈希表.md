```cpp
#include <iostream>  
#include <iomanip>  
using namespace std;  
  
struct list {  
    int pre;  
    int key;  
    int next;  
};  
  
int main() {  
    int begin, n;  
    cin >> begin >> n;  
    list sheet[100001];  // 创建地址数组哈希表
    int key[10000]={0};  
    for (int i = 0; i < n; i++) {   // 将输入节点存储并映射到哈希表上
        int pre, key, next;  
        cin >> pre >> key >> next;  
        list mid;  
        mid.pre = pre;  
        mid.key = key;  
        mid.next = next;  
        sheet[pre] = mid;  
    }  
    list* List=(list*)malloc(sizeof(list)*n);   // 根据哈希表，将输入数据以静态链表的形式存储起来
    int cur = begin;  
    int i = 0;  
    while (cur != -1) {  
        List[i] = sheet[cur];  
        cur = List[i].next;  
        i++;  
    }  
    list* newList = (list*)malloc(sizeof(list)*n);  // 开始筛选不需要删除和需要删除的节点，分别新建静态链表存储
    list* deleList = (list*)malloc(sizeof(list)*n);  
    int newcount = 0, delecount = 0;  
    cur = begin; i = 0;  
    while (cur != -1) {  
        int IkeyI = List[i].key > 0 ? List[i].key : -List[i].key;  
        if (key[IkeyI] == 0) {  
            newList[newcount] = List[i];  
            cur = List[i].next;  
            newcount++;  
            key[IkeyI] = 1;  
        }  
        else {  
            deleList[delecount] = List[i];  
            cur = List[i].next;  
            delecount++;  
        }  
        i++;  
    }  
    setfill('0');  // 最后输出
    for (int i = 0; i < newcount; i++) {  
        cout << setfill('0') << setw(5) << newList[i].pre << " " << newList[i].key << " ";  
        if (i+1 != newcount) cout << setfill('0') << setw(5) << newList[i+1].pre << endl;  
        else cout << -1 << endl;  
    }  
    for (int i = 0; i < delecount; i++) {  
        cout << setfill('0') << setw(5) << deleList[i].pre << " " << deleList[i].key << " ";  
        if (i+1 != delecount) cout << setfill('0') << setw(5) << deleList[i+1].pre << endl;  
        else cout << -1 << endl;  
    }  
    return 0;  
}
```

>[!warning]
>静态链表由于数组存储的关系，每个节点存储的next地址数据若非主动是**不变的**。因此最后输出每个节点的next时，需要用下一个节点的pre表示！！

>[!tip]
>静态链表最好不要用数组的逻辑遍历，借助间接的指针cur，使用`cur = List[i].next`的方式移动指针遍历。

>[!tip]- 如何输出前导零？
>[[C++-Learning/1-笔记/STL/C++输出前导零|C++输出前导零]]笔记中记录了两种方法：使用`<iomanip>`中的`<< setfill('0') << setw(5)`函数，注意只对**后一个**数字生效。
