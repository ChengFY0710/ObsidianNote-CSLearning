**二叉树的遍历（Traversal）**：按某条搜索路径巡访树中每个结点，使得每个结点均被访问一次，而且仅被访问一次。

## 01 先序遍历
若二叉树为空，则空操作；
否则（1）访问**根结点**；（2）先序遍历左子树；（3）先序遍历右子树。
![[引用文件/DataStructure-Learning/Pasted image 20261013210643.png|259]]
==🟢**先序遍历的递归算法**：==
```cpp
Status PreOrder(BiTree T, Status(* Visit)(TElemType e)){
	// 采用二叉链表存储结构，Visit是对数据元素操作的应用函数。
	// 先序遍历二叉树T的递归算法，对每个数据元素调用函数Visit。
	if(T){ // 非空树,
		if(Visit(T->data)){
			if(PreOrder(T->Lchild, Visit)){
				if(PreOrder(T->Rchild, Visit)){
					return OK;
				}
			}
		}
		
		return ERROR;
	}
	else return OK;
}
```
==🟢**先序遍历的非递归算法：**==
```cpp
Status PreOrderTraverse(BiTree T, Status(* Visit)(TElemType e)){
	// 采用二叉链表存储结构，借助栈实现先序遍历的非递归算法。
	InitStack(S);
	if(T) Push(S, T);
	while(!StackEmpty(S)){
		Pop(S, p);
		if(!Visit(p->data)) return ERROR;
		if(p->Rchild) Push(S, p->Rchild);  // 右孩子先入栈
		if(p->Lchild) Push(S, p->Lchild);  // 左孩子后入栈，先出栈访问
	}
	return OK;
}
```

## 02 中序遍历
若二叉树为空，则空操作；
否则，（1）中序遍历左子树；（2）访问根结点；（3）中序遍历右子树。
![[引用文件/DataStructure-Learning/Pasted image 20261013210837.png|256]]
==🟢**中序遍历的递归算法：**==
```cpp
Status InOrder(BiTree T, Status(* Visit)(TElemType e)){
	//采用二叉链表存储结构，Visit是对数据元素操作的应用函数。
	//中序遍历二叉树T的递归算法，对每个数据元素调用函数Visit。
	if(T){
		if(InOrder(T->Lchild, Visit)){
			if(Visit(T->data)){
				if(InOrder(T->Rchild, Visit))
					return OK;
			}
		}
		return ERROR;
	}
	else return OK;
}
```

==🟢**中序遍历的非递归算法 1 ：**==
```cpp
Status InOrderTraverse(BiTree T, Status(* Visit)(TElemType e)){
	//采用二叉链表存储结构，Visit是对数据元素操作的应用函数。
	InitStack(S); Push(S,T);  //根指针进栈
	while(!StackEmpty(S)){
		while(GetTop(S, p) && p) 
			Push(S, p->Lchild);  //向左走到尽头
		Pop(S, p);  //空指针退栈
		if(!StackEmpty(S)){  //访问结点，向右一步
			Pop(S, p);
			if(!Visit(p->data)) return ERROR;
			Push(S, p->Rchild);
		} //if
	} // while
	return OK;	
} // InOrderTraverse
```

==🟢**中序遍历的非递归算法 2 ：**==
```cpp
Status InOrderTraverse(BiTree T, Status(* Visit)(TElemType e)){
	// 采用二叉链表存储结构，Visit是对数据元素操作的应用函数。
	InitStack(S); p = T;
	while(p || !StackEmpty(S)){
		if(p){Push(S, p); p = p->Lchild;} // 根指针进栈，遍历左子树。
		else{  // 根指针退栈，访问根结点，遍历右子树。
			Pop(S, p);
			if(!Visit(p->data)) return ERROR;
			p = p->Rchild;
		}
	}
	return OK;
}
```
## 03 后序遍历
若二叉树为空，则空操作；
否则，（1）后序遍历左子树；（2）后序遍历右子树；（3）访问根结点。
![[引用文件/DataStructure-Learning/Pasted image 20261013211042.png|264]]
==🟢**后序遍历的递归算法：**==
```cpp
Status PostOrder(BiTree T, Status(* Visit)(TElemType e)){
	// 采用二叉链表存储结构，Visit是对数据元素操作的应用函数。
	// 后序遍历二叉树T的递归算法，对每个数据元素调用函数Visit。
	if(T){
		if(PostOrder(T->Lchild, Visit)){
			if(PostOrder(T->Rchild, Visit)){
				if(Visit(T->data))
					return OK;
			}
		}
		return ERROR;
	}
	else return OK;
}
```
==🟢**后序遍历的非递归算法：**==
```cpp
Status PostOrderTraverse(BiTree T, Status(* Visit)(TElemType e)){
	// 采用二叉链表存储结构，借助栈实现后序遍历的非递归算法。
	InitStack(S);
	p = T; pre = NULL;  // pre 指向最近访问过的结点
	while(p || !StackEmpty(S)){
		if(p){  // 向左走到尽头
			Push(S, p);
			p = p->Lchild;
		}
		else{
			GetTop(S, p);
			if(p->Rchild && p->Rchild != pre)  // 右子树未访问，转向右子树
				p = p->Rchild;
			else{  // 右子树已访问或为空，访问根结点
				Pop(S, p);
				if(!Visit(p->data)) return ERROR;
				pre = p;
				p = NULL;
			}
		}
	}
	return OK;
}
```

## 04 层次遍历
若二叉树为空，则空操作；
否则，从**根结点**开始，**自上而下、从左到右**依次访问每一层的结点。
==🟢**层次遍历的队列算法：**==
```cpp
Status LevelOrder(BiTree T, Status(* Visit)(TElemType e)){
	// 采用二叉链表存储结构，Visit是对数据元素操作的应用函数。
	// 层次遍历二叉树T，借助队列实现。
	InitQueue(Q);
	if(T) EnQueue(Q, T);  // 根指针入队
	while(!QueueEmpty(Q)){
		DeQueue(Q, p);  // 队头结点出队并访问
		if(!Visit(p->data)) return ERROR;
		if(p->Lchild) EnQueue(Q, p->Lchild);  // 左孩子入队
		if(p->Rchild) EnQueue(Q, p->Rchild);  // 右孩子入队
	}
	return OK;
}
```

```cpp
// 具体代码
void FloorPrint_QUEUE(pTreeNode &Tree) //层序遍历_队列实现
{
    queue < pTreeNode> q;
    if (Tree != NULL)
    {
        q.push(Tree);   //根节点进队列
    }

    while (q.empty() == false)  //队列不为空判断
    {
        cout << q.front()->data << " → "; 

        if (q.front()->leftPtr != NULL)   //如果有左孩子，leftChild入队列
        {
            q.push(q.front()->leftPtr);   
        }

        if (q.front()->rightPtr != NULL)   //如果有右孩子，rightChild入队列
        {
            q.push(q.front()->rightPtr);
        }
        q.pop();  //已经遍历过的节点出队列
    }
}
```

>[!tip]
>先序、中序、后序遍历本质上都是**深度优先搜索（DFS）**，递归工作栈的深度为树的高度；层次遍历是**广度优先搜索（BFS）**，需要借助**队列**实现。

==🟢**层次遍历的数组算法：**==
```cpp
void FloorPrint(pTreeNode Tree)  //层序遍历
{
    pTreeNode temp[100];   //创建pTreeNode指针类型的指针数组
    int in = 0;
    int out = 0;

    temp[in++] = Tree;  //先保存二叉树根节点 

    while (in > out)
    {
        if (temp[out])
        {
            cout << temp[out]->data << " → ";
            temp[in++] = temp[out]->leftPtr;
            temp[in++] = temp[out]->rightPtr;
        }
        out++;
    }
}
```

## 05 遍历序列小结
以各遍历示意图中的二叉树为例：

| 遍历方式 | 访问根结点的时机 | 遍历序列 |
| --- | --- | --- |
| 先序遍历 | 左、右子树之前 | ABDCEGFHI |
| 中序遍历 | 左、右子树之间 | DBAEGCHFI |
| 后序遍历 | 左、右子树之后 | DBGEHIFCA |
| 层次遍历 | 按层次自上而下、从左到右 | ABCDEFGHI |

>[!warning]
>+ 已知**先序序列和中序序列**，或**后序序列和中序序列**，可以唯一确定一棵二叉树。
>+ 已知**先序序列和后序序列**，不能唯一确定一棵二叉树。

## 06 由遍历序列确定二叉树
### （1）先序 + 中序
+ 先序序列的**第一个结点**一定是根结点。
+ 在中序序列中找到根结点，根结点**左边**是左子树的中序序列，**右边**是右子树的中序序列。
+ 根据左子树结点个数，在先序序列中划分出左、右子树的先序序列，再递归构造。

### （2）后序 + 中序
+ 后序序列的**最后一个结点**一定是根结点。
+ 在中序序列中找到根结点并划分左、右子树，再递归构造，方法同上。

### （3）层次 + 中序
+ 层次序列的**第一个结点**是根结点。
+ 在中序序列中划分左、右子树，再按层次顺序确定各子树的根结点，递归构造。

### （4）先序 + 后序
+ **不能**唯一确定一棵二叉树。
+ 因为无法区分只有一个孩子时，该孩子是**左孩子**还是**右孩子**。

以遍历示意图中的二叉树为例：
+ 先序序列：`ABDCEGFHI`
+ 中序序列：`DBAEGCHFI`
```
        A
       / \
      B   C
     /   / \
    D   E   F
         \  / \
          G H I
```
由先序序列的第一个结点 `A` 可知根为 `A`；在中序序列中 `A` 左边为 `DB`，右边为 `EGCHFI`，继续递归划分即可还原整棵二叉树。


```cpp
// 根据后序和中序输出前序
#include <iostream>
using namespace std;

int post[30];
int in[30];


void preOrder(int postL, int postR, int inL, int inR) {
    if (postL > postR) return;

    int root = post[postR];
    static bool first = true;
    if (first) {
        cout << root;
        first = false;
    } else { cout << " " << root; } // 行末不能有空格。

    int k;
    for (k = inL; k <= inR; k++) {
        if (in[k] == root) break;
    }

    int numLeft = k - inL;

    // 遍历左子树
    preOrder(postL, postL + numLeft - 1, inL, k - 1);
    // 遍历右子树
    preOrder(postL + numLeft, postR - 1, k + 1, inR);
}

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> post[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> in[i];
    }

    cout << "Preorder: ";

    preOrder(0, n-1, 0, n-1);

    return 0;
}
```
